\vspace{4pt}
\noindent\textbf{\color{azulprof}Objetivo.} Determinar cómo depende la longitud
de onda de los modos estacionarios de una cuerda, excitada por un motor, de la
tensión $T$, de la densidad lineal $\mu$ y de la distancia $L$ entre los puntos
de apoyo; y, a partir de ello, estimar $\mu$ y compararla con la medida directa
con la báscula, siguiendo el guion de la práctica.

% =========================================================================
\section{Lo esencial de la teoría}
% =========================================================================
En una cuerda tensa las ondas viajan a $v=\sqrt{T/\mu}$ y, para una frecuencia
$f$ fija (la del motor), su longitud de onda es $\lambda=v/f$. Con los dos
extremos fijos a distancia $L$, solo caben modos estacionarios con un número
entero $n$ de vientres (antinodos):
\begin{equation}\label{eq:modos}
  \lambda_n=\frac{2L}{n},\qquad n=1,2,3,\dots
\end{equation}
Combinando ambas relaciones,
\begin{equation}\label{eq:lineal}
  \lambda^{2}=\frac{1}{\mu f^{2}}\;T
  \qquad\Longrightarrow\qquad
  \boxed{\;\lambda^{2}\ \text{frente a}\ T\ \text{es una recta de pendiente}\ a=\frac{1}{\mu f^{2}}\;}
\end{equation}

\begin{consejo}
\small\textbf{Tres ideas para entender lo que veremos.}
(i)~Como $T_n=\mu f^{2}(2L/n)^{2}$, cuantos \emph{más} vientres, \emph{menos}
tensión: el modo $n=1$ es el que más tensión necesita.
(ii)~La pendiente solo depende de $\mu$ (y de $f$): al doblar la cuerda la
pendiente se reduce a la mitad, mientras que cambiar $L$ \emph{no} cambia la
pendiente, solo desplaza los puntos a lo largo de la misma recta
(\cref{fig:grafica}).
(iii)~De la pendiente se despeja la densidad: $\mu=1/(a f^{2})$.
\end{consejo}

\section{Procedimiento}
% =========================================================================
\begin{enumerate}
  \item \textbf{Preparación.} Fijar la varilla en la posición $L_1$ (la más
    lejana) y medir $L_1$ con la regla. Atar la cuerda \emph{simple} a la
    excéntrica y al dinamómetro, y comprobar que este marca \emph{cero} sin
    tensión.
  \item \textbf{Frecuencia.} Encender el motor y medir su frecuencia $f$ con el
    estroboscopio. Se mide una vez y vale para todas las series (el motor no
    cambia de velocidad), aunque conviene comprobarla al final.
  \item \textbf{Barrido en tensión.} Partir de una tensión alta y desplazar
    \emph{lentamente} el soporte del dinamómetro hasta que aparezca la onda
    estacionaria más nítida y amplia con $n=1$ vientre. Anotar $T$. Seguir
    aflojando para $n=2,3,4$ y, si se puede, $5$
    (\cref{fig:modos}).
  \item \textbf{Cuerda doble.} Repetir el barrido con la cuerda doble
    (densidad $2\mu$) y la misma $L_1$.
  \item \textbf{Segunda distancia.} Pasar la varilla a $L_2$, medirla y repetir
    con la cuerda simple y con la doble.
  \item \textbf{Densidad.} Pesar un trozo de cuerda de longitud $\ell$ conocida
    (cuanto más largo, mejor) en la báscula: $\mu=m/\ell$.
\end{enumerate}

\begin{consejo}
\small\textbf{Consejos prácticos.}
El estroboscopio también deja la cuerda inmóvil a $f/2$, $f/3$\ldots: la
frecuencia buena es la \emph{más alta} a la que se ve una sola imagen quieta
(al doble se ven dos). Cada tensión conviene leerla en el punto de máxima
amplitud del modo, acercándonos desde ambos lados; si da tiempo, repetir
cada lectura dos o tres veces. Anotar la resolución del dinamómetro, de la
regla y de la báscula: son las incertidumbres de tipo~B.
\end{consejo}

\begin{caja}{Qué hay que hacer antes de irse}
\begin{enumerate}[leftmargin=1.5em]
  \item Calcular $\lambda_n=2L/n$ y $\lambda_n^{2}$ para cada medida.
  \item Representar en una misma gráfica $\lambda^{2}$ frente a $T$ para las
    cuatro series ($L_1$ y $L_2$; cuerda simple y doble) y hacer un ajuste
    lineal de cada una, con sus pendientes $a$ e incertidumbres.
  \item Comparar las pendientes: (i)~misma $L$ y distinta densidad (deben
    estar en razón $2:1$); (ii)~distinta $L$ y misma densidad (deben coincidir).
  \item Estimar la densidad de cada pendiente, $\mu=1/(a f^{2})$, propagando
    la incertidumbre:
    $\dfrac{u(\mu)}{\mu}=\sqrt{\Bigl(\dfrac{u(a)}{a}\Bigr)^{2}+\Bigl(\dfrac{2\,u(f)}{f}\Bigr)^{2}}$,
    y compararla con la de la báscula, $\mu=m/\ell$, mediante sus intervalos
    al \SI{95}{\percent}.
\end{enumerate}
\end{caja}
